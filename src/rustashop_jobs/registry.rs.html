54/// Job status returned to the admin UI.
55#[derive(Debug, Clone, Copy, Serialize, Deserialize, ToSchema, PartialEq, Eq)]
56#[serde(rename_all = "snake_case")]
57pub enum SandboxJobStatus {
58    /// Runner still working.
59    Running,
60    /// Guest finished; host validated adjustments (quote path).
61    Succeeded,
62    /// Guest proposal validated; waiting for host commit.
63    AwaitingCommit,
64    /// Host applied the proposal to commerce state.
65    Committed,
66    /// Operator discarded the proposal without mutating commerce.
67    Discarded,
68    /// Guest or validation failed.
69    Failed,
70}

152/// In-process job + audit store shared by HTTP and the runner.
153#[derive(Clone, Default)]
154pub struct SandboxJobRegistry {

158#[derive(Default)]
159struct RegistryInner {
160    jobs: HashMap<String, JobRecord>,
161    audit: Vec<SandboxAuditRecord>,
162}

164impl std::fmt::Debug for SandboxJobRegistry {
165    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
166        let guard = self.inner.lock().expect("sandbox registry mutex");
167        f.debug_struct("SandboxJobRegistry")
168            .field("job_count", &guard.jobs.len())
169            .field("audit_count", &guard.audit.len())
170            .finish()
171    }

181    /// Inserts a running job and an audit stub; returns the public view.
182    ///
183    /// # Panics
184    ///
185    /// Panics if the mutex is poisoned.
186    #[must_use]
187    pub fn start_job(&self, job_type: &str, source_hash: &str, actor: &str) -> SandboxJobResponse {
188        let id = new_job_id();
189        let created_at_unix = unix_now();
190        let response = SandboxJobResponse {
191            id: id.clone(),
192            job_type: job_type.to_owned(),
193            status: SandboxJobStatus::Running,
194            source_hash: source_hash.to_owned(),
195            adjustments: None,
196            proposal: None,
197            error: None,
198        };
199        let audit = SandboxAuditRecord {
200            job_id: id,
201            actor: actor.to_owned(),
202            job_type: job_type.to_owned(),
203            source_hash: source_hash.to_owned(),
204            status: "running".to_owned(),
205            created_at_unix,
206        };
207        let mut guard = self.inner.lock().expect("sandbox registry mutex");
208        guard.jobs.insert(
209            response.id.clone(),
210            JobRecord {
211                response: response.clone(),
212            },
213        );
214        guard.audit.push(audit);
215        drop(guard);
216        status_store::persist_job(&response);
217        response
218    }
219
220    /// Updates job + matching audit row after a terminal quote/failure outcome.
221    ///
222    /// # Panics
223    ///
224    /// Panics if the mutex is poisoned.
225    pub fn finish_job(
226        &self,
227        job_id: &str,
228        status: SandboxJobStatus,
229        adjustments: Option<Vec<Adjustment>>,
230        error: Option<String>,
231    ) {
232        let mut guard = self.inner.lock().expect("sandbox registry mutex");
233        if let Some(job) = guard.jobs.get_mut(job_id) {
234            job.response.status = status;
235            job.response.adjustments = adjustments.map(|rows| {
236                rows.into_iter()
237                    .map(SandboxAdjustmentResponse::from)
238                    .collect()
239            });
240            job.response.error = error;
241        }
242        Self::touch_audit(&mut guard, job_id, status_label(status));
243        let snapshot = guard.jobs.get(job_id).map(|job| job.response.clone());
244        drop(guard);
245        if let Some(response) = snapshot {
246            status_store::persist_job(&response);
247        }
248    }
249
250    /// Stores a validated proposal and moves the job to [`SandboxJobStatus::AwaitingCommit`].
251    ///
252    /// # Panics
253    ///
254    /// Panics if the mutex is poisoned.
255    pub fn set_awaiting_commit(&self, job_id: &str, proposal: SandboxProposalResponse) {
256        let mut guard = self.inner.lock().expect("sandbox registry mutex");
257        if let Some(job) = guard.jobs.get_mut(job_id) {
258            job.response.status = SandboxJobStatus::AwaitingCommit;
259            job.response.proposal = Some(proposal);
260            job.response.error = None;
261        }
262        Self::touch_audit(&mut guard, job_id, "awaiting_commit");
263        let snapshot = guard.jobs.get(job_id).map(|job| job.response.clone());
264        drop(guard);
265        if let Some(response) = snapshot {
266            status_store::persist_job(&response);
267        }
268    }
269
270    /// Marks a proposal job as committed or discarded.
271    ///
272    /// # Panics
273    ///
274    /// Panics if the mutex is poisoned.
275    pub fn finalize_proposal(&self, job_id: &str, status: SandboxJobStatus) {
276        let mut guard = self.inner.lock().expect("sandbox registry mutex");
277        if let Some(job) = guard.jobs.get_mut(job_id) {
278            job.response.status = status;
279        }
280        Self::touch_audit(&mut guard, job_id, status_label(status));
281        let snapshot = guard.jobs.get(job_id).map(|job| job.response.clone());
282        drop(guard);
283        if let Some(response) = snapshot {
284            status_store::persist_job(&response);
285        }
286    }

288    /// Test helper: awaiting commit with no proposal payload (covers commit guard).
289    ///
290    /// # Panics
291    ///
292    /// Panics if the mutex is poisoned.
293    pub fn set_awaiting_commit_without_proposal(&self, job_id: &str) {
294        let mut guard = self.inner.lock().expect("sandbox registry mutex");
295        if let Some(job) = guard.jobs.get_mut(job_id) {
296            job.response.status = SandboxJobStatus::AwaitingCommit;
297            job.response.proposal = None;
298        }
299        Self::touch_audit(&mut guard, job_id, "awaiting_commit");
300        drop(guard);
301    }
302
303    fn touch_audit(guard: &mut RegistryInner, job_id: &str, status_label: &str) {
304        if let Some(row) = guard
305            .audit
306            .iter_mut()
307            .rev()
308            .find(|row| row.job_id == job_id)
309        {
310            status_label.clone_into(&mut row.status);
311        }
312    }

314    /// Looks up a job by id.
315    ///
316    /// # Panics
317    ///
318    /// Panics if the mutex is poisoned.
319    #[must_use]
320    pub fn get(&self, job_id: &str) -> Option<SandboxJobResponse> {
321        let memory = self
322            .inner
323            .lock()
324            .expect("sandbox registry mutex")
325            .jobs
326            .get(job_id)
327            .map(|job| job.response.clone());
328        if let Some(job) = memory {
329            return Some(job);
330        }
331        status_store::load_job(job_id)
332    }

334    /// Newest-first audit rows (capped).
335    ///
336    /// # Panics
337    ///
338    /// Panics if the mutex is poisoned.
339    #[must_use]
340    pub fn list_audit(&self, limit: usize) -> Vec<SandboxAuditRecord> {
341        let guard = self.inner.lock().expect("sandbox registry mutex");
342        guard.audit.iter().rev().take(limit).cloned().collect()
343    }

346const fn status_label(status: SandboxJobStatus) -> &'static str {
347    match status {
348        SandboxJobStatus::Running => "running",
349        SandboxJobStatus::Succeeded => "succeeded",
350        SandboxJobStatus::AwaitingCommit => "awaiting_commit",
351        SandboxJobStatus::Committed => "committed",
352        SandboxJobStatus::Discarded => "discarded",
353        SandboxJobStatus::Failed => "failed",
354    }
355}

357fn new_job_id() -> String {
358    let mut bytes = [0_u8; 16];
359    getrandom::fill(&mut bytes).expect("getrandom");
360    bytes
361        .iter()
362        .fold(String::with_capacity(32), |mut out, byte| {
363            use std::fmt::Write as _;
364            let _ = write!(out, "{byte:02x}");
365            out
366        })
367}
368
369fn unix_now() -> u64 {
370    std::time::SystemTime::now()
371        .duration_since(std::time::UNIX_EPOCH)
372        .unwrap_or_else(|_| std::time::Duration::from_secs(0))
373        .as_secs()
374}

376/// Stable fingerprint of guest source for audit.
377#[must_use]
378pub fn source_hash(source: &str) -> String {
379    let mut hasher = std::collections::hash_map::DefaultHasher::new();
380    source.hash(&mut hasher);
381    format!("{:016x}", hasher.finish())
382}

398        let job = registry.start_job(JOB_TYPE_QUOTE, "hash", "admin-bearer");
399        assert_eq!(job.status, SandboxJobStatus::Running);
400        assert!(format!("{registry:?}").contains("job_count"));
401        registry.finish_job(&job.id, SandboxJobStatus::Succeeded, Some(vec![]), None);
402        let done = registry.get(&job.id).expect("job");
403        assert_eq!(done.status, SandboxJobStatus::Succeeded);
404        let audit = registry.list_audit(10);
405        assert_eq!(audit.len(), 1);

422    #[test]
423    fn registry_awaiting_commit_and_finalize() {
424        let registry = SandboxJobRegistry::new();
425        let job = registry.start_job(JOB_TYPE_CART_QUANTITY, "hash", "admin-bearer");
426        registry.set_awaiting_commit(
427            &job.id,
428            SandboxProposalResponse {
429                event_type: "cart.line_quantity_proposed".into(),
430                cart_id: "c1".into(),
431                product_id: "v1".into(),
432                quantity: 2,
433                operator: "set".into(),
434            },
435        );
436        let waiting = registry.get(&job.id).expect("job");
437        assert_eq!(waiting.status, SandboxJobStatus::AwaitingCommit);
438        assert!(waiting.proposal.is_some());
439        registry.finalize_proposal(&job.id, SandboxJobStatus::Committed);
440        assert_eq!(
441            registry.get(&job.id).expect("job").status,
442            SandboxJobStatus::Committed
443        );
444        assert_eq!(registry.list_audit(1)[0].status, "committed");
445        registry.finalize_proposal(&job.id, SandboxJobStatus::Discarded);
446        assert_eq!(
447            registry.get(&job.id).expect("job").status,
448            SandboxJobStatus::Discarded
449        );
450        assert_eq!(registry.list_audit(1)[0].status, "discarded");
451        registry.finalize_proposal(&job.id, SandboxJobStatus::Running);
452        assert_eq!(registry.list_audit(1)[0].status, "running");
453        registry.finalize_proposal(&job.id, SandboxJobStatus::AwaitingCommit);
454        assert_eq!(registry.list_audit(1)[0].status, "awaiting_commit");
455    }
