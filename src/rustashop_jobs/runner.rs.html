11/// Runs the Python quote guest and updates registry + hub.
12pub async fn run_quote_job(
13    registry: &SandboxJobRegistry,
14    hub: &SandboxJobHub,
15    job_id: &str,
16    cart: &CartSnapshot,
17    source: &str,
18) {
19    hub.publish(&SandboxJobEvent::log(
20        job_id,
21        "starting Wasmer Python quote",
22    ));
23    match invoke_python_quote(cart, source).await {
24        Ok(raw) => match apply_validated_adjustments(cart, raw) {
25            Ok(applied) => {
26                hub.publish(&SandboxJobEvent::log(
27                    job_id,
28                    format!("validated {} adjustment(s)", applied.len()),
29                ));
30                registry.finish_job(job_id, SandboxJobStatus::Succeeded, Some(applied), None);
31                hub.publish(&SandboxJobEvent::finished(job_id, "ok"));
32            }
33            Err(error) => {
34                let message = format!("validation failed: {error:#}");
35                hub.publish(&SandboxJobEvent::log(job_id, &message));
36                registry.finish_job(job_id, SandboxJobStatus::Failed, None, Some(message));
37                hub.publish(&SandboxJobEvent::finished(job_id, "error"));
38            }
39        },
40        Err(error) => {
41            let message = format!("guest failed: {error:#}");
42            hub.publish(&SandboxJobEvent::log(job_id, &message));
43            registry.finish_job(job_id, SandboxJobStatus::Failed, None, Some(message));
44            hub.publish(&SandboxJobEvent::finished(job_id, "error"));
45        }
46    }
47}

49/// Runs the PHP cart-quantity guest and stores an awaiting-commit proposal.
50pub async fn run_cart_quantity_job(
51    registry: &SandboxJobRegistry,
52    hub: &SandboxJobHub,
53    job_id: &str,
54    input: &LegacyHookInput,
55    source: &str,
56) {
57    hub.publish(&SandboxJobEvent::log(
58        job_id,
59        "starting Wasmer PHP cart_quantity guest",
60    ));
61    match invoke_php_migration_hook(input, source).await {
62        Ok(raw) => match accept_validated_domain_event(raw) {
63            Ok(draft) => {
64                let proposal = proposal_from_draft(&draft);
65                hub.publish(&SandboxJobEvent::log(
66                    job_id,
67                    format!(
68                        "validated proposal {} on cart {}",
69                        proposal.event_type, proposal.cart_id
70                    ),
71                ));
72                registry.set_awaiting_commit(job_id, proposal.clone());
73                hub.publish(&SandboxJobEvent::proposal(
74                    job_id,
75                    SandboxProposalEventBody {
76                        event_type: proposal.event_type,
77                        cart_id: proposal.cart_id,
78                        product_id: proposal.product_id,
79                        quantity: proposal.quantity,
80                        operator: proposal.operator,
81                    },
82                ));
83            }
84            Err(error) => {
85                let message = format!("validation failed: {error:#}");
86                hub.publish(&SandboxJobEvent::log(job_id, &message));
87                registry.finish_job(job_id, SandboxJobStatus::Failed, None, Some(message));
88                hub.publish(&SandboxJobEvent::finished(job_id, "error"));
89            }
90        },
91        Err(error) => {
92            let message = format!("guest failed: {error:#}");
93            hub.publish(&SandboxJobEvent::log(job_id, &message));
94            registry.finish_job(job_id, SandboxJobStatus::Failed, None, Some(message));
95            hub.publish(&SandboxJobEvent::finished(job_id, "error"));
96        }
97    }
98}

100fn proposal_from_draft(draft: &DomainEventDraft) -> SandboxProposalResponse {
101    SandboxProposalResponse {
102        event_type: draft.event_type.clone(),
103        cart_id: draft.cart_id.clone(),
104        product_id: draft.product_id.clone(),
105        quantity: draft.quantity,
106        operator: draft.operator.clone(),
107    }
108}
