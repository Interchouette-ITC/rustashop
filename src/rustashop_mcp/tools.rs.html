1//! Commerce AI tool input schemas (MCP / agents).
2//!
3//! The stable catalog (`TOOLS`) lives in `rustashop-domain` so Actix and Axum
4//! share one definition.
5
6use schemars::JsonSchema;
7use serde::{Deserialize, Serialize};
8
9pub use rustashop_domain::{
10    TOOLS, ToolDescriptor, ToolEffect, ToolScope, tool_by_name, tools_catalog_json,
11};
12
13/// Input for `list_products`.
14#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
15pub struct ListProductsInput {
16    /// Maximum rows (commerce caps at 100).
17    #[serde(skip_serializing_if = "Option::is_none")]
18    pub limit: Option<u32>,
19    /// Rows to skip.
20    #[serde(skip_serializing_if = "Option::is_none")]
21    pub offset: Option<u32>,
22}
23
24/// Input for `get_product`.
25#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
26pub struct GetProductInput {
27    /// Product id.
28    pub id: String,
29}
30
31/// Input for `create_cart`.
32#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
33pub struct CreateCartInput {
34    /// ISO currency (default `EUR` when omitted).
35    #[serde(skip_serializing_if = "Option::is_none")]
36    pub currency: Option<String>,
37}
38
39/// Input for `get_cart`.
40#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
41pub struct GetCartInput {
42    /// Cart id.
43    pub id: String,
44}
45
46/// Input for `add_cart_line`.
47#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
48pub struct AddCartLineInput {
49    /// Cart id (path `{id}`).
50    pub cart_id: String,
51    /// Variant to add.
52    pub variant_id: String,
53    /// Quantity greater than zero.
54    pub quantity: i32,
55}
56
57/// Input for `update_cart_line` / `delete_cart_line` path params.
58#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
59pub struct CartLineRefInput {
60    /// Cart id.
61    pub cart_id: String,
62    /// Line id.
63    pub line_id: String,
64    /// Replacement quantity (update only).
65    #[serde(skip_serializing_if = "Option::is_none")]
66    pub quantity: Option<i32>,
67}
68
69/// Input for `place_order`.
70#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
71pub struct PlaceOrderInput {
72    /// Cart to check out.
73    pub cart_id: String,
74    /// Idempotency key (also sent as `Idempotency-Key` on HTTP).
75    pub idempotency_key: String,
76}
77
78/// Input for admin list tools (`list_admin_products` / `list_admin_orders`).
79#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
80pub struct AdminListInput {
81    /// Maximum rows.
82    #[serde(skip_serializing_if = "Option::is_none")]
83    pub limit: Option<u32>,
84    /// Rows to skip.
85    #[serde(skip_serializing_if = "Option::is_none")]
86    pub offset: Option<u32>,
87}
88
89/// Input for `patch_order_status`.
90#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, JsonSchema)]
91pub struct PatchOrderStatusInput {
92    /// Order id.
93    pub id: String,
94    /// Target status (`pending`, `paid`, `shipped`, `cancelled`, …).
95    pub status: String,
96}
