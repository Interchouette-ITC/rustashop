1//! Commerce AI tool catalog shared by MCP and in-app agent UIs.
2//!
3//! Tools are host-mediated: no SQL-from-prompt, integer money only, and
4//! commit-class effects require stronger policy than cart draft writes.
5
6use serde::Serialize;
7
8/// Who may invoke the tool.
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
10#[serde(rename_all = "snake_case")]
11pub enum ToolScope {
12    /// Customer / shop session surface (catalog, cart, checkout).
13    Shop,
14    /// Operator surface (admin bearer + opaque URI prefix).
15    Admin,
16}
17
18/// Side effect class for policy gates.
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
20#[serde(rename_all = "snake_case")]
21pub enum ToolEffect {
22    /// Pure read of commerce state.
23    Read,
24    /// Mutates draft session state (open cart lines). Not payment capture.
25    DraftWrite,
26    /// Places an order or changes order lifecycle; host-authorized commit.
27    Commit,
28}
29
30/// One callable tool shared by MCP and in-app agents.
31#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
32pub struct ToolDescriptor {
33    /// Stable MCP / agent function name (`snake_case`).
34    pub name: &'static str,
35    /// One-line capability summary.
36    pub summary: &'static str,
37    /// Matching `OpenAPI` path template.
38    pub openapi_path: &'static str,
39    /// HTTP method for the commerce route.
40    pub method: &'static str,
41    /// Shop vs admin authz class.
42    pub scope: ToolScope,
43    /// Read / draft write / commit.
44    pub effect: ToolEffect,
45    /// Autonomous agents must not run this without human or strict policy approve.
46    pub human_approve_for_autonomous: bool,
47}

152/// Looks up a tool by stable name.
153#[must_use]
154pub fn tool_by_name(name: &str) -> Option<&'static ToolDescriptor> {
155    TOOLS.iter().find(|tool| tool.name == name)
156}
157
158/// Serializes the tool catalog for MCP `list_tools` / agent discovery.
159///
160/// # Errors
161///
162/// Returns [`serde_json::Error`] when serialization fails.
163pub fn tools_catalog_json() -> Result<serde_json::Value, serde_json::Error> {
164    serde_json::to_value(TOOLS)
165}
