1//! Commerce repository ports. Adapters live in persist crates.

3use serenade_contracts::{EntityId, PageRequest, RepositoryError};
4use std::future::Future;
5
6/// Catalog product reads. Product writes live beside admin use-cases.
7pub trait ProductRepository: Send + Sync {
8    /// Error type for this adapter.
9    type Error: RepositoryError;
10    /// Product identifier type (for example UUID string).
11    type Id: EntityId;
12    /// Product aggregate or view model.
13    type Product: Send + Sync;
14
15    /// Load one product by primary key.
16    fn find_by_id(
17        &self,
18        id: &Self::Id,
19    ) -> impl Future<Output = Result<Option<Self::Product>, Self::Error>> + Send;
20
21    /// Load one product by URL slug.
22    fn find_by_slug(
23        &self,
24        slug: &str,
25    ) -> impl Future<Output = Result<Option<Self::Product>, Self::Error>> + Send;
26
27    /// List products for storefront browse.
28    fn list(
29        &self,
30        page: PageRequest,
31    ) -> impl Future<Output = Result<Vec<Self::Product>, Self::Error>> + Send;
32}

34/// Category tree reads for navigation and catalog filters.
35pub trait CategoryRepository: Send + Sync {
36    /// Error type for this adapter.
37    type Error: RepositoryError;
38    /// Category identifier type.
39    type Id: EntityId;
40    /// Category model.
41    type Category: Send + Sync;
42
43    /// Load one category by primary key.
44    fn find_by_id(
45        &self,
46        id: &Self::Id,
47    ) -> impl Future<Output = Result<Option<Self::Category>, Self::Error>> + Send;
48
49    /// Load one category by slug within an optional parent scope.
50    fn find_by_slug(
51        &self,
52        slug: &str,
53        parent_id: Option<&Self::Id>,
54    ) -> impl Future<Output = Result<Option<Self::Category>, Self::Error>> + Send;
55
56    /// List child categories for a parent (`None` = roots).
57    fn list_children(
58        &self,
59        parent_id: Option<&Self::Id>,
60        page: PageRequest,
61    ) -> impl Future<Output = Result<Vec<Self::Category>, Self::Error>> + Send;
62}

64/// Cart session persistence. Line mutations snapshot price in the domain layer.
65pub trait CartRepository: Send + Sync {
66    /// Error type for this adapter.
67    type Error: RepositoryError;
68    /// Cart identifier type.
69    type Id: EntityId;
70    /// Cart aggregate.
71    type Cart: Send + Sync;
72
73    /// Resolve a cart by opaque session token.
74    fn find_by_token(
75        &self,
76        token: &str,
77    ) -> impl Future<Output = Result<Option<Self::Cart>, Self::Error>> + Send;
78
79    /// Insert or update a cart aggregate.
80    fn save(&self, cart: &Self::Cart) -> impl Future<Output = Result<(), Self::Error>> + Send;
81
82    /// Remove a cart when checkout completes or session expires.
83    fn delete(&self, id: &Self::Id) -> impl Future<Output = Result<(), Self::Error>> + Send;
84}
