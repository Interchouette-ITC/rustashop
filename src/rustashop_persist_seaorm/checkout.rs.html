24fn parse_uuid(id: &str) -> Result<Uuid, PersistenceError> {
25    Uuid::parse_str(id).map_err(|_| PersistenceError::InvalidInput {
26        message: format!("invalid id `{id}`"),

30fn is_unique_violation(error: &DbErr) -> bool {
31    error.to_string().contains("duplicate key")

34fn money(amount_minor: i64, currency: &str) -> Result<Money, PersistenceError> {
35    let currency = Currency::new(currency).map_err(|error| PersistenceError::InvalidInput {
36        message: error.to_string(),
37    })?;
38    Ok(Money::new(amount_minor, currency))

41pub(crate) fn order_from_models(
42    model: commerce_order::Model,
43    lines: Vec<order_line::Model>,
44) -> Result<Order, PersistenceError> {
45    let code = model.currency.clone();
46    let currency = Currency::new(&code).map_err(|error| PersistenceError::InvalidInput {
47        message: error.to_string(),
48    })?;
49    let mut order_lines = Vec::with_capacity(lines.len());
50    for line in lines {
51        order_lines.push(OrderLine {
52            id: line.id.to_string(),
53            order_id: line.order_id.to_string(),
54            variant_id: line.variant_id.map(|id| id.to_string()),
55            quantity: line.quantity,
56            unit_price: money(line.unit_price_minor, &line.currency)?,
57            line_total: money(line.line_total_minor, &line.currency)?,
58            product_name: line.product_name,
59            variant_sku: line.variant_sku,
60        });
61    }
62    Ok(Order {
63        id: model.id.to_string(),
64        number: model.number,
65        cart_id: model.cart_id.map(|id| id.to_string()),
66        customer_id: model.customer_id.map(|id| id.to_string()),
67        state: model.state,
68        payment_status: PAYMENT_STATUS_PENDING.to_owned(),
69        items_total: money(model.items_total_minor, &code)?,
70        total: money(model.total_minor, &code)?,
71        currency,
72        idempotency_key: model.idempotency_key,
73        lines: order_lines,
74    })
75}

77impl SeaOrmCatalogRepository {
78    /// Places an order from `cart_id`. Replays when `idempotency_key` already exists.
79    ///
80    /// # Errors
81    ///
82    /// Returns [`PersistenceError::NotFound`], [`PersistenceError::InvalidInput`],
83    /// [`PersistenceError::Conflict`], or an internal adapter error.
84    pub async fn checkout_cart(
85        &self,
86        cart_id: &str,
87        idempotency_key: Option<&str>,
88    ) -> Result<Order, PersistenceError> {
89        if let Some(key) = idempotency_key
90            && let Some(existing) = find_order_by_key(&self.db, key).await?
91        {
92            return Ok(existing);
93        }
94        let txn = self.db.begin().await.map_err(|error| internal(&error))?;
95        match checkout_in_txn(&txn, cart_id, idempotency_key).await {
96            Ok(order) => {
97                txn.commit().await.map_err(|error| internal(&error))?;
98                Ok(order)
99            }
100            Err(error) => {
101                txn.rollback().await.ok();
102                if let Some(key) = idempotency_key
103                    && matches!(
104                        error,
105                        PersistenceError::Conflict {
106                            constraint: "idempotency_key"
107                        }
108                    )
109                {
110                    return find_order_by_key(&self.db, key).await?.ok_or(error);
111                }
112                Err(error)
113            }
114        }
115    }
116}

118async fn checkout_in_txn<C: ConnectionTrait>(
119    db: &C,
120    cart_id: &str,
121    idempotency_key: Option<&str>,
122) -> Result<Order, PersistenceError> {
123    let uuid = parse_uuid(cart_id)?;
124    let model = cart::Entity::find_by_id(uuid)
125        .lock_exclusive()
126        .one(db)
127        .await
128        .map_err(|error| internal(&error))?
129        .ok_or_else(|| PersistenceError::NotFound {
130            entity: "cart",
131            id: cart_id.to_owned(),
132        })?;
133    let lines = cart_line::Entity::find()
134        .filter(cart_line::Column::CartId.eq(uuid))
135        .order_by_asc(cart_line::Column::CreatedAt)
136        .order_by_asc(cart_line::Column::Id)
137        .all(db)
138        .await
139        .map_err(|error| internal(&error))?;
140    let cart = cart_from_models(model, lines)?;
141    if cart.status == CartStatus::CheckedOut {
142        if let Some(key) = idempotency_key
143            && let Some(existing) = find_order_by_key(db, key).await?
144        {
145            return Ok(existing);
146        }
147        return Err(PersistenceError::Conflict {
148            constraint: "cart_status",
149        });
150    }
151    cart.ensure_checkoutable()
152        .map_err(|error| PersistenceError::InvalidInput {
153            message: error.to_string(),
154        })?;
155    let items_total = cart
156        .items_total()
157        .map_err(|error| PersistenceError::InvalidInput {
158            message: error.to_string(),
159        })?;
160    let now = chrono::Utc::now().into();
161    let order_id = Uuid::new_v4();
162    let number = format!("RS-{}", &order_id.simple().to_string()[..12]);
163    let inserted = commerce_order::ActiveModel {
164        id: Set(order_id),
165        number: Set(number),
166        customer_id: Set(match &cart.customer_id {
167            Some(id) => Some(parse_uuid(id)?),
168            None => None,
169        }),
170        cart_id: Set(Some(uuid)),
171        state: Set("placed".to_owned()),
172        currency: Set(cart.currency.as_str().to_owned()),
173        items_total_minor: Set(items_total.amount_minor),
174        total_minor: Set(items_total.amount_minor),
175        idempotency_key: Set(idempotency_key.map(ToOwned::to_owned)),
176        created_at: Set(now),
177        updated_at: Set(now),
178        placed_at: Set(Some(now)),
179    }
180    .insert(db)
181    .await
182    .map_err(|error| {
183        if is_unique_violation(&error) {
184            PersistenceError::Conflict {
185                constraint: "idempotency_key",
186            }
187        } else {
188            internal(&error)
189        }
190    })?;
191    insert_order_lines(db, inserted.id, &cart, now).await?;
192    mark_cart_checked_out(db, uuid, &cart.id, now).await?;
193    load_order(db, inserted.id).await
194}

196async fn insert_order_lines<C: ConnectionTrait>(
197    db: &C,
198    order_id: Uuid,
199    cart: &Cart,
200    now: DateTimeWithTimeZone,
201) -> Result<(), PersistenceError> {
202    for line in &cart.lines {
203        let line_total = line
204            .line_total()
205            .expect("line totals already validated by items_total");
206        order_line::ActiveModel {
207            id: Set(Uuid::new_v4()),
208            order_id: Set(order_id),
209            variant_id: Set(Some(parse_uuid(&line.variant_id)?)),
210            quantity: Set(line.quantity),
211            unit_price_minor: Set(line.unit_price.amount_minor),
212            line_total_minor: Set(line_total.amount_minor),
213            currency: Set(line.unit_price.currency.as_str().to_owned()),
214            product_name: Set(line.product_name.clone()),
215            variant_sku: Set(line.variant_sku.clone()),
216            created_at: Set(now),
217            updated_at: Set(now),
218        }
219        .insert(db)
220        .await
221        .map_err(|error| internal(&error))?;
222    }
223    Ok(())
224}

226pub(crate) async fn mark_cart_checked_out<C: ConnectionTrait>(
227    db: &C,
228    cart_uuid: Uuid,
229    cart_id: &str,
230    now: DateTimeWithTimeZone,
231) -> Result<(), PersistenceError> {
232    let mut active: cart::ActiveModel = cart::Entity::find_by_id(cart_uuid)
233        .one(db)
234        .await
235        .map_err(|error| internal(&error))?
236        .ok_or_else(|| PersistenceError::NotFound {
237            entity: "cart",
238            id: cart_id.to_owned(),
239        })?
240        .into();
241    active.status = Set(CartStatus::CheckedOut.as_str().to_owned());
242    active.updated_at = Set(now);
243    active.update(db).await.map_err(|error| internal(&error))?;
244    Ok(())
245}

247pub(crate) async fn find_order_by_key<C: ConnectionTrait>(
248    db: &C,
249    key: &str,
250) -> Result<Option<Order>, PersistenceError> {
251    let model = commerce_order::Entity::find()
252        .filter(commerce_order::Column::IdempotencyKey.eq(key))
253        .one(db)
254        .await
255        .map_err(|error| internal(&error))?;
256    let Some(model) = model else {
257        return Ok(None);
258    };
259    let lines = load_order_lines(db, model.id).await?;
260    Ok(Some(order_from_models(model, lines)?))
261}

263pub(crate) async fn load_order<C: ConnectionTrait>(
264    db: &C,
265    order_id: Uuid,
266) -> Result<Order, PersistenceError> {
267    let model = commerce_order::Entity::find_by_id(order_id)
268        .one(db)
269        .await
270        .map_err(|error| internal(&error))?
271        .ok_or_else(|| PersistenceError::NotFound {
272            entity: "order",
273            id: order_id.to_string(),
274        })?;
275    let lines = load_order_lines(db, order_id).await?;
276    order_from_models(model, lines)
277}

279pub(crate) async fn load_order_lines<C: ConnectionTrait>(
280    db: &C,
281    order_id: Uuid,
282) -> Result<Vec<order_line::Model>, PersistenceError> {
283    order_line::Entity::find()
284        .filter(order_line::Column::OrderId.eq(order_id))
285        .order_by_asc(order_line::Column::CreatedAt)
286        .order_by_asc(order_line::Column::Id)
287        .all(db)
288        .await
289        .map_err(|error| internal(&error))
290}
