23/// Applies pending `SeaORM` migrations against the given connection.
24///
25/// # Errors
26///
27/// Returns [`sea_orm::DbErr`] when migration execution fails.
28pub async fn migrate(connection: &DatabaseConnection) -> Result<(), sea_orm::DbErr> {
29    migration::Migrator::up(connection, None).await
30}
31
32/// Inserts the catalog seed rows (`ON CONFLICT DO NOTHING`).
33///
34/// # Errors
35///
36/// Returns [`sea_orm::DbErr`] when the script fails.
37pub async fn seed_catalog(connection: &DatabaseConnection) -> Result<(), sea_orm::DbErr> {
38    connection.execute_unprepared(CATALOG_SEED_SQL).await?;
39    Ok(())
40}

42/// Connects with `DATABASE_URL` and returns a catalog repository.
43///
44/// # Errors
45///
46/// Returns [`MigrateError`] when the URL is missing or the database is unreachable.
47pub async fn catalog_from_env() -> Result<SeaOrmCatalogRepository, MigrateError> {
48    let url = require_database_url(std::env::var("DATABASE_URL"))?;
49    let mut options = ConnectOptions::new(url);
50    options.max_connections(5);
51    let connection = Database::connect(options).await?;
52    Ok(SeaOrmCatalogRepository::new(connection))
53}
54
55/// Connects with `DATABASE_URL` and runs pending migrations.
56///
57/// # Errors
58///
59/// Returns [`MigrateError`] when the database is unreachable or migrations fail.
60pub async fn migrate_from_env() -> Result<(), MigrateError> {
61    let url = require_database_url(std::env::var("DATABASE_URL"))?;
62    let connection = Database::connect(&url).await?;
63    migrate(&connection).await?;
64    Ok(())
65}
66
67fn require_database_url(
68    result: Result<String, std::env::VarError>,
69) -> Result<String, MigrateError> {
70    result.map_err(|_| MigrateError::MissingDatabaseUrl)
71}
72
73/// Migration runner errors.
74#[derive(Debug, thiserror::Error)]
75pub enum MigrateError {
76    /// `DATABASE_URL` is not set.
77    #[error("DATABASE_URL must be set")]
78    MissingDatabaseUrl,
79    /// Underlying `SeaORM` database error.
80    #[error(transparent)]
81    Database(#[from] sea_orm::DbErr),
82}
