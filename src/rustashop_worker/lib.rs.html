1//! Serenade console ops + sandbox messenger worker commands.

3mod commands;

5use std::sync::Arc;
6
7use serenade_bundle::{BundleError, COMMAND_TAG, Extension};
8use serenade_cache::ArrayAdapter;
9use serenade_config::Config;
10use serenade_console::CommandService;
11use serenade_di::{ContainerBuilder, ServiceDefinition};
12
13use commands::{AboutCommand, CacheClearCommand, MessengerConsumeCommand};
14
15/// Service id for the ops catalog cache pool (`cache:clear`).
16pub const CATALOG_CACHE_POOL_SERVICE: &str = "rustashop.cache.catalog";
17
18/// DI extension registering rustashop console commands.
19#[derive(Clone, Copy, Debug, Default)]
20pub struct WorkerExtension;
21
22impl Extension for WorkerExtension {
23    fn alias(&self) -> &'static str {
24        "rustashop_worker"
25    }
26
27    fn load(&self, config: &Config, builder: &mut ContainerBuilder) -> Result<(), BundleError> {
28        config.apply_to(builder.parameters_mut());
29        builder.register(ServiceDefinition::new(CATALOG_CACHE_POOL_SERVICE), |_| {
30            Ok(Box::new(ArrayAdapter::new()))
31        })?;
32        builder
33            .register(
34                ServiceDefinition::new("console.command.rustashop_about").with_tag(COMMAND_TAG),
35                |_| Ok(Box::new(CommandService(Arc::new(AboutCommand)))),
36            )
37            .expect("rustashop about command id is unique");
38        builder
39            .register(
40                ServiceDefinition::new("console.command.cache_clear").with_tag(COMMAND_TAG),
41                |_| Ok(Box::new(CommandService(Arc::new(CacheClearCommand)))),
42            )
43            .expect("cache clear command id is unique");
44        builder
45            .register(
46                ServiceDefinition::new("console.command.messenger_consume").with_tag(COMMAND_TAG),
47                |_| Ok(Box::new(CommandService(Arc::new(MessengerConsumeCommand)))),
48            )
49            .expect("messenger consume command id is unique");
50        Ok(())
51    }
52}

54#[cfg(test)]
55mod tests {
56    use super::*;
57    use serenade_console::{Command, Input};
58    use serenade_kernel::Environment;
59
60    #[test]
61    fn worker_extension_registers_commands_and_cache_pool() {
62        assert_eq!(WorkerExtension.alias(), "rustashop_worker");
63        let config = Config::empty();
64        let mut builder = ContainerBuilder::new();
65        builder
66            .parameters_mut()
67            .set("rustashop.name", "coverage-shop");
68        WorkerExtension
69            .load(&config, &mut builder)
70            .expect("load worker extension");
71        let container = std::sync::Arc::new(builder.compile().expect("compile"));
72
73        let about = container
74            .get_as::<CommandService>("console.command.rustashop_about")
75            .expect("about");
76        assert_eq!(about.0.name(), "rustashop:about");
77        assert_ne!(about.0.description(), "");
78
79        let cache = container
80            .get_as::<CommandService>("console.command.cache_clear")
81            .expect("cache");
82        assert_eq!(cache.0.name(), "cache:clear");
83        assert_ne!(cache.0.description(), "");
84
85        let consume = container
86            .get_as::<CommandService>("console.command.messenger_consume")
87            .expect("consume");
88        assert_eq!(consume.0.name(), "messenger:consume");
89        assert_ne!(consume.0.description(), "");
90
91        let input = Input::new(
92            Environment::Dev,
93            true,
94            Vec::new(),
95            Some(std::sync::Arc::clone(&container)),
96        );
97        AboutCommand.execute(&input).expect("about with container");
98        CacheClearCommand.execute(&input).expect("cache clear");
99
100        let limit_input = Input::new(
101            Environment::Dev,
102            false,
103            vec!["--limit".into(), "1".into()],
104            Some(container),
105        );
106        MessengerConsumeCommand
107            .execute(&limit_input)
108            .expect("consume limit");
109    }
