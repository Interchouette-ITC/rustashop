50/// Default bind address when `RUSTASHOP_BIND` is unset.
51pub const DEFAULT_BIND: &str = "127.0.0.1:8080";
52
53/// Environment variable for the API listen address.
54pub const BIND_ENV: &str = "RUSTASHOP_BIND";
55
56/// Returns the bind address from `RUSTASHOP_BIND` or [`DEFAULT_BIND`].
57#[must_use]
58pub fn bind_address() -> String {
59    std::env::var(BIND_ENV).unwrap_or_else(|_| DEFAULT_BIND.to_owned())
60}
61
62/// Registers HTTP routes on `cfg` (admin prefix from env / local default).
63pub fn routes(cfg: &mut web::ServiceConfig) {
64    configure_app(cfg, &AdminApiPrefix::from_env());
65}
66
67/// Registers test-only Actix extras (`Swagger` UI, install static files).
68///
69/// Production binds via Serenade `listen` only. JSON commerce routes use
70/// [`configure_serenade_front`] (or the kernel matcher under listen).
71pub fn configure_routes(cfg: &mut web::ServiceConfig) {
72    cfg.service(swagger_ui());
73    install_routes::configure_install_from_env(cfg);
74}
75
76/// Registers Serenade front routes then test-only Actix extras (integration tests).
77pub fn configure_app(cfg: &mut web::ServiceConfig, admin_prefix: &AdminApiPrefix) {
78    configure_serenade_front(cfg, admin_prefix.as_str());
79    configure_routes(cfg);
80}
