60/// Job status returned to the admin UI.
61#[derive(Debug, Clone, Copy, Serialize, Deserialize, ToSchema, PartialEq, Eq)]
62#[serde(rename_all = "snake_case")]
63pub enum SandboxJobStatus {
64    /// Runner still working.
65    Running,
66    /// Guest finished; host validated adjustments (quote path).
67    Succeeded,
68    /// Guest proposal validated; waiting for host commit.
69    AwaitingCommit,
70    /// Host applied the proposal to commerce state.
71    Committed,
72    /// Operator discarded the proposal without mutating commerce.
73    Discarded,
74    /// Guest or validation failed.
75    Failed,
76}

187    /// Inserts a running job and an audit stub; returns the public view.
188    ///
189    /// # Panics
190    ///
191    /// Panics if the mutex is poisoned.
192    #[must_use]
193    pub fn start_job(&self, job_type: &str, source_hash: &str, actor: &str) -> SandboxJobResponse {
194        let id = new_job_id();
195        let created_at_unix = unix_now();
196        let response = SandboxJobResponse {
197            id: id.clone(),
198            job_type: job_type.to_owned(),
199            status: SandboxJobStatus::Running,
200            source_hash: source_hash.to_owned(),
201            adjustments: None,
202            proposal: None,
203            error: None,
204        };
205        let audit = SandboxAuditRecord {
206            job_id: id,
207            actor: actor.to_owned(),
208            job_type: job_type.to_owned(),
209            source_hash: source_hash.to_owned(),
210            status: "running".to_owned(),
211            created_at_unix,
212        };
213        let mut guard = self.inner.lock().expect("sandbox registry mutex");
214        guard.jobs.insert(
215            response.id.clone(),
216            JobRecord {
217                response: response.clone(),
218            },
219        );
220        guard.audit.push(audit);
221        response
222    }
223
224    /// Updates job + matching audit row after a terminal quote/failure outcome.
225    ///
226    /// # Panics
227    ///
228    /// Panics if the mutex is poisoned.
229    pub fn finish_job(
230        &self,
231        job_id: &str,
232        status: SandboxJobStatus,
233        adjustments: Option<Vec<Adjustment>>,
234        error: Option<String>,
235    ) {
236        let mut guard = self.inner.lock().expect("sandbox registry mutex");
237        if let Some(job) = guard.jobs.get_mut(job_id) {
238            job.response.status = status;
239            job.response.adjustments = adjustments.map(|rows| {
240                rows.into_iter()
241                    .map(SandboxAdjustmentResponse::from)
242                    .collect()
243            });
244            job.response.error = error;
245        }
246        Self::touch_audit(&mut guard, job_id, status_label(status));
247        drop(guard);
248    }
249
250    /// Stores a validated proposal and moves the job to [`SandboxJobStatus::AwaitingCommit`].
251    ///
252    /// # Panics
253    ///
254    /// Panics if the mutex is poisoned.
255    pub fn set_awaiting_commit(&self, job_id: &str, proposal: SandboxProposalResponse) {
256        let mut guard = self.inner.lock().expect("sandbox registry mutex");
257        if let Some(job) = guard.jobs.get_mut(job_id) {
258            job.response.status = SandboxJobStatus::AwaitingCommit;
259            job.response.proposal = Some(proposal);
260            job.response.error = None;
261        }
262        Self::touch_audit(&mut guard, job_id, "awaiting_commit");
263        drop(guard);
264    }
265
266    /// Marks a proposal job as committed or discarded.
267    ///
268    /// # Panics
269    ///
270    /// Panics if the mutex is poisoned.
271    pub fn finalize_proposal(&self, job_id: &str, status: SandboxJobStatus) {
272        let mut guard = self.inner.lock().expect("sandbox registry mutex");
273        if let Some(job) = guard.jobs.get_mut(job_id) {
274            job.response.status = status;
275        }
276        Self::touch_audit(&mut guard, job_id, status_label(status));
277        drop(guard);
278    }

296    fn touch_audit(guard: &mut RegistryInner, job_id: &str, status_label: &str) {
297        if let Some(row) = guard
298            .audit
299            .iter_mut()
300            .rev()
301            .find(|row| row.job_id == job_id)
302        {
303            status_label.clone_into(&mut row.status);
304        }
305    }

308    ///
309    /// # Panics
310    ///
311    /// Panics if the mutex is poisoned.
312    #[must_use]

314        self.inner

322    /// Newest-first audit rows (capped).
323    ///
324    /// # Panics

334const fn status_label(status: SandboxJobStatus) -> &'static str {
335    match status {
336        SandboxJobStatus::Running => "running",
337        SandboxJobStatus::Succeeded => "succeeded",
338        SandboxJobStatus::AwaitingCommit => "awaiting_commit",
339        SandboxJobStatus::Committed => "committed",
340        SandboxJobStatus::Discarded => "discarded",
341        SandboxJobStatus::Failed => "failed",
342    }
343}

345fn new_job_id() -> String {
346    let mut bytes = [0_u8; 16];
347    getrandom::fill(&mut bytes).expect("getrandom");
348    bytes
349        .iter()
350        .fold(String::with_capacity(32), |mut out, byte| {
351            use std::fmt::Write as _;
352            let _ = write!(out, "{byte:02x}");
353            out
354        })
355}

357fn unix_now() -> u64 {
358    std::time::SystemTime::now()
359        .duration_since(std::time::UNIX_EPOCH)
360        .unwrap_or_else(|_| std::time::Duration::from_secs(0))
361        .as_secs()
362}

364/// Stable fingerprint of guest source for audit.
365#[must_use]
366pub fn source_hash(source: &str) -> String {
367    let mut hasher = std::collections::hash_map::DefaultHasher::new();
368    source.hash(&mut hasher);
369    format!("{:016x}", hasher.finish())
370}

372fn cart_from_request(body: &CreateSandboxJobRequest) -> Option<CartSnapshot> {
373    let currency = body.currency.as_ref()?.clone();
374    let lines = body.lines.as_ref()?;
375    Some(CartSnapshot {
376        currency: currency.clone(),
377        lines: lines
378            .iter()
379            .map(|line| CartLine {
380                sku: line.sku.clone(),
381                quantity: line.quantity,
382                unit_price: Money {
383                    amount_minor: line.unit_price_minor,
384                    currency: currency.clone(),
385                },
386            })
387            .collect(),
388    })
389}

391/// Creates a quote or autonomous cart-quantity job and returns the running job JSON.
392pub fn create_sandbox_job_response(
393    auth: &AdminAuthConfig,
394    bearer: Option<&str>,
395    registry: &SandboxJobRegistry,
396    hub: &SandboxJobHub,
397    body: &[u8],
398) -> Response {
399    if let Err(error) = auth.authorize_bearer(bearer) {
400        return api_error_json_response(&error);
401    }
402    let request: CreateSandboxJobRequest = match serde_json::from_slice(body) {
403        Ok(value) => value,
404        Err(_) => {
405            return api_error_json_response(&ApiError::Unprocessable(
406                "invalid create sandbox job body".into(),
407            ));
408        }
409    };
410    match request.job_type.as_str() {
411        JOB_TYPE_QUOTE => create_quote_job(registry, hub, &request),
412        JOB_TYPE_CART_QUANTITY => {
413            crate::sandbox_autonomous::create_cart_quantity_job(registry, hub, &request)
414        }
415        _ => api_error_json_response(&ApiError::Unprocessable(
416            "unsupported job_type (use quote or cart_quantity)".into(),
417        )),
418    }
419}
420
421fn create_quote_job(
422    registry: &SandboxJobRegistry,
423    hub: &SandboxJobHub,
424    request: &CreateSandboxJobRequest,
425) -> Response {
426    let Some(cart) = cart_from_request(request) else {
427        return api_error_json_response(&ApiError::Unprocessable(
428            "currency and at least one line are required".into(),
429        ));
430    };
431    if cart.currency.trim().is_empty() || cart.lines.is_empty() {
432        return api_error_json_response(&ApiError::Unprocessable(
433            "currency and at least one line are required".into(),
434        ));
435    }
436
437    let source = quote_fixture_source();
438    let hash = source_hash(source);
439    let job = registry.start_job(JOB_TYPE_QUOTE, &hash, "admin-bearer");
440    let job_id = job.id.clone();
441    let registry_runner = registry.clone();
442    let hub_runner = hub.clone();
443
444    tokio::spawn(async move {
445        run_quote_job(&registry_runner, &hub_runner, &job_id, &cart, source).await;
446    });
447
448    json_response(202, &job)
449}

522async fn run_quote_job(
523    registry: &SandboxJobRegistry,
524    hub: &SandboxJobHub,
525    job_id: &str,
526    cart: &CartSnapshot,
527    source: &str,
528) {
529    hub.publish(&SandboxJobEvent::log(
530        job_id,
531        "starting Wasmer Python quote",
532    ));
533    match invoke_python_quote(cart, source).await {
534        Ok(raw) => match apply_validated_adjustments(cart, raw) {
535            Ok(applied) => {
536                hub.publish(&SandboxJobEvent::log(
537                    job_id,
538                    format!("validated {} adjustment(s)", applied.len()),
539                ));
540                registry.finish_job(job_id, SandboxJobStatus::Succeeded, Some(applied), None);
541                hub.publish(&SandboxJobEvent::finished(job_id, "ok"));
542            }
543            Err(error) => {
544                let message = format!("validation failed: {error:#}");
545                hub.publish(&SandboxJobEvent::log(job_id, &message));
546                registry.finish_job(job_id, SandboxJobStatus::Failed, None, Some(message));
547                hub.publish(&SandboxJobEvent::finished(job_id, "error"));
548            }
549        },
550        Err(error) => {
551            let message = format!("guest failed: {error:#}");
552            hub.publish(&SandboxJobEvent::log(job_id, &message));
553            registry.finish_job(job_id, SandboxJobStatus::Failed, None, Some(message));
554            hub.publish(&SandboxJobEvent::finished(job_id, "error"));
555        }
556    }
557}

742                unit_price: Money {
743                    amount_minor: 5000,
744                    currency: "EUR".into(),
745                },
746            }],
747        };
748
749        let guest_fail = registry.start_job(JOB_TYPE_QUOTE, "bad", "admin");
