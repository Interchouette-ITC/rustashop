15/// Builds the production Actix app: cart WS route + Serenade kernel catch-all.
16#[must_use]
17pub fn commerce_app(
18    kernel: web::Data<AsyncHttpKernel>,
19    hub: web::Data<CartHub>,
20    catalog: web::Data<CatalogRepository>,
21) -> App<
22    impl ServiceFactory<
23        ServiceRequest,
24        Config = (),
25        Response = ServiceResponse<impl MessageBody>,
26        Error = Error,
27        InitError = (),
28    >,
29> {
30    App::new()
31        .app_data(kernel)
32        .app_data(hub)
33        .app_data(catalog)
34        .route("/v1/carts/{id}/ws", web::get().to(cart_ws))
35        .default_service(web::to(kernel_service))
36}

38/// Bound commerce server plus the addresses it listens on.
39pub struct BoundCommerce {
40    /// Actix server future (run with [`serenade_http_actix::await_bound`]).
41    pub server: actix_web::dev::Server,
42    /// Bound socket addresses.
43    pub addrs: Vec<SocketAddr>,
44}
45
46/// Binds `addr` with commerce HTTP + cart WebSocket (does not await).
47///
48/// # Errors
49///
50/// Propagates bind errors.
51pub fn bind_commerce_server(
52    addr: impl ToSocketAddrs,
53    kernel: AsyncHttpKernel,
54    hub: CartHub,
55    catalog: CatalogRepository,
56) -> std::io::Result<BoundCommerce> {
57    let kernel = web::Data::new(kernel);
58    let hub = web::Data::new(hub);
59    let catalog = web::Data::new(catalog);
60    let http = HttpServer::new(move || commerce_app(kernel.clone(), hub.clone(), catalog.clone()))
61        .bind(addr)?;
62    let addrs = http.addrs();
63    Ok(BoundCommerce {
64        server: http.run(),
65        addrs,
66    })
67}
