18/// Shared Actix app data for commerce listen.
19#[derive(Clone)]
20pub struct CommerceListenData {
21    /// Serenade HTTP kernel.
22    pub kernel: web::Data<AsyncHttpKernel>,
23    /// Cart push hub.
24    pub cart_hub: web::Data<CartHub>,
25    /// Catalog for cart WS auth.
26    pub catalog: web::Data<CatalogRepository>,
27    /// Sandbox job push hub.
28    pub sandbox_hub: web::Data<SandboxJobHub>,
29    /// Admin bearer for sandbox WS.
30    pub admin_auth: web::Data<AdminAuthConfig>,
31    /// Operator API path segment (sandbox WS path).
32    pub admin_prefix: String,
33}
34
35/// Builds the production Actix app: WS routes + Serenade kernel catch-all.
36#[must_use]
37pub fn commerce_app(
38    data: CommerceListenData,
39) -> App<
40    impl ServiceFactory<
41        ServiceRequest,
42        Config = (),
43        Response = ServiceResponse<impl MessageBody>,
44        Error = Error,
45        InitError = (),
46    >,
47> {
48    let sandbox_ws_path = format!("/v1/{}/sandbox/jobs/{{id}}/ws", data.admin_prefix);
49    App::new()
50        .app_data(data.kernel)
51        .app_data(data.cart_hub)
52        .app_data(data.catalog)
53        .app_data(data.sandbox_hub)
54        .app_data(data.admin_auth)
55        .route("/v1/carts/{id}/ws", web::get().to(cart_ws))
56        .route(&sandbox_ws_path, web::get().to(sandbox_job_ws))
57        .default_service(web::to(kernel_service))
58}

60/// Bound commerce server plus the addresses it listens on.
61pub struct BoundCommerce {
62    /// Actix server future (run with [`serenade_http_actix::await_bound`]).
63    pub server: actix_web::dev::Server,
64    /// Bound socket addresses.
65    pub addrs: Vec<SocketAddr>,
66}

68/// Binds `addr` with commerce HTTP + WebSocket routes (does not await).
69///
70/// # Errors
71///
72/// Propagates bind errors.
73pub fn bind_commerce_server(
74    addr: impl ToSocketAddrs,
75    kernel: AsyncHttpKernel,
76    cart_hub: CartHub,
77    catalog: CatalogRepository,
78    sandbox_hub: SandboxJobHub,
79    admin_auth: AdminAuthConfig,
80    admin_prefix: impl Into<String>,
81) -> std::io::Result<BoundCommerce> {
82    let data = CommerceListenData {
83        kernel: web::Data::new(kernel),
84        cart_hub: web::Data::new(cart_hub),
85        catalog: web::Data::new(catalog),
86        sandbox_hub: web::Data::new(sandbox_hub),
87        admin_auth: web::Data::new(admin_auth),
88        admin_prefix: admin_prefix.into(),
89    };
90    let http = HttpServer::new(move || commerce_app(data.clone())).bind(addr)?;
91    let addrs = http.addrs();
92    Ok(BoundCommerce {
93        server: http.run(),
94        addrs,
95    })
96}
