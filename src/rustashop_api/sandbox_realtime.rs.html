6use serde::Serialize;

9const CHANNEL_CAPACITY: usize = 64;
10
11/// Proposal payload embedded in `job.proposal` events.
12#[derive(Debug, Clone, Serialize, PartialEq, Eq)]
13pub struct SandboxProposalEventBody {
14    /// Domain event type (for example `cart.line_quantity_proposed`).
15    pub event_type: String,
16    /// Target cart id.
17    pub cart_id: String,
18    /// Legacy `id_product` mapped to rustashop variant id for v0.
19    pub product_id: String,
20    /// Proposed quantity operand.
21    pub quantity: u32,
22    /// Operator (`up` / `down` / `set`).
23    pub operator: String,
24}
25
26/// Server-pushed sandbox job event (JSON over WebSocket).
27#[derive(Debug, Clone, Serialize, PartialEq, Eq)]
28pub struct SandboxJobEvent {
29    /// Stable event name (`job.log` / `job.proposal` / `job.finished`).
30    #[serde(rename = "type")]
31    pub event_type: String,
32    /// Protocol version.
33    pub version: u32,
34    /// Job id this event belongs to.
35    pub job_id: String,
36    /// Log line for `job.log`; empty otherwise.
37    #[serde(skip_serializing_if = "String::is_empty")]
38    pub message: String,
39    /// Exit status string for `job.finished` (`ok` / `error` / `committed` / `discarded`).
40    #[serde(skip_serializing_if = "Option::is_none")]
41    pub status: Option<String>,
42    /// Validated guest proposal for `job.proposal`.
43    #[serde(skip_serializing_if = "Option::is_none")]
44    pub proposal: Option<SandboxProposalEventBody>,
45}
46
47impl SandboxJobEvent {
48    /// Builds a `job.log` v1 event.
49    #[must_use]
50    pub fn log(job_id: impl Into<String>, message: impl Into<String>) -> Self {
51        Self {
52            event_type: "job.log".to_owned(),
53            version: 1,
54            job_id: job_id.into(),
55            message: message.into(),
56            status: None,
57            proposal: None,
58        }
59    }
60
61    /// Builds a `job.proposal` v1 event (awaiting host commit).
62    #[must_use]
63    pub fn proposal(job_id: impl Into<String>, proposal: SandboxProposalEventBody) -> Self {
64        Self {
65            event_type: "job.proposal".to_owned(),
66            version: 1,
67            job_id: job_id.into(),
68            message: String::new(),
69            status: None,
70            proposal: Some(proposal),
71        }
72    }
73
74    /// Builds a `job.finished` v1 event.
75    #[must_use]
76    pub fn finished(job_id: impl Into<String>, status: impl Into<String>) -> Self {
77        Self {
78            event_type: "job.finished".to_owned(),
79            version: 1,
80            job_id: job_id.into(),
81            message: String::new(),
82            status: Some(status.into()),
83            proposal: None,
84        }
85    }
86}

88/// In-process fan-out of sandbox job events by job id.
89#[derive(Clone, Default)]
90pub struct SandboxJobHub {
91    rooms: Arc<Mutex<HashMap<String, broadcast::Sender<String>>>>,
92}
93
94impl std::fmt::Debug for SandboxJobHub {
95    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
96        let rooms = self.rooms.lock().expect("sandbox hub mutex");
97        f.debug_struct("SandboxJobHub")
98            .field("room_count", &rooms.len())
99            .finish()
100    }
101}
102
103impl SandboxJobHub {
104    /// Creates an empty hub.
105    #[must_use]
106    pub fn new() -> Self {
107        Self::default()
108    }
109
110    /// Subscribes to push events for `job_id`.
111    ///
112    /// # Panics
113    ///
114    /// Panics if the hub mutex is poisoned.
115    #[must_use]
116    pub fn subscribe(&self, job_id: &str) -> broadcast::Receiver<String> {
117        let mut rooms = self.rooms.lock().expect("sandbox hub mutex");
118        if let Some(sender) = rooms.get(job_id) {
119            return sender.subscribe();
120        }
121        let (sender, receiver) = broadcast::channel(CHANNEL_CAPACITY);
122        rooms.insert(job_id.to_owned(), sender);
123        receiver
124    }
125
126    /// Publishes a job event (no-op if nobody is subscribed).
127    ///
128    /// # Panics
129    ///
130    /// Panics if the hub mutex is poisoned or the event fails to serialize.
131    pub fn publish(&self, event: &SandboxJobEvent) {
132        let payload = serde_json::to_string(event).expect("sandbox job event serializes");
133        let rooms = self.rooms.lock().expect("sandbox hub mutex");
134        if let Some(sender) = rooms.get(&event.job_id) {
135            let _ = sender.send(payload);
136        }
137    }
138}

140#[cfg(test)]
141mod tests {
142    use super::{SandboxJobEvent, SandboxJobHub, SandboxProposalEventBody};
143
144    #[test]
145    fn publish_reaches_subscriber() {
146        let hub = SandboxJobHub::new();
147        assert!(format!("{hub:?}").contains("room_count"));
148        let mut rx = hub.subscribe("job-1");
149        let mut rx2 = hub.subscribe("job-1");
150        hub.publish(&SandboxJobEvent::log("job-1", "hello"));
151        let raw = rx.try_recv().expect("event");
152        let parsed: serde_json::Value = serde_json::from_str(&raw).expect("json");
153        assert_eq!(parsed["type"], "job.log");
154        assert_eq!(parsed["message"], "hello");
155        assert!(rx2.try_recv().is_ok());
156        hub.publish(&SandboxJobEvent::log("nobody", "ignored"));
157    }
158
159    #[test]
160    fn finished_event_includes_status() {
161        let event = SandboxJobEvent::finished("j", "ok");
162        let raw = serde_json::to_string(&event).expect("json");
163        assert!(raw.contains("job.finished"));
164        assert!(raw.contains("\"ok\""));
165    }
166
167    #[test]
168    fn proposal_event_includes_body() {
169        let event = SandboxJobEvent::proposal(
170            "j",
171            SandboxProposalEventBody {
172                event_type: "cart.line_quantity_proposed".into(),
173                cart_id: "c1".into(),
174                product_id: "v1".into(),
175                quantity: 2,
176                operator: "set".into(),
177            },
178        );
179        let raw = serde_json::to_string(&event).expect("json");
180        assert!(raw.contains("job.proposal"));
181        assert!(raw.contains("cart.line_quantity_proposed"));
182    }
