20impl Modify for AdminSecurityAddon {
21    fn modify(&self, openapi: &mut utoipa::openapi::OpenApi) {
22        if let Some(components) = openapi.components.as_mut() {
23            components.add_security_scheme(
24                "admin_bearer",
25                SecurityScheme::Http(
26                    HttpBuilder::new()
27                        .scheme(HttpAuthScheme::Bearer)
28                        .bearer_format("token")
29                        .build(),
30                ),
31            );
32        }
33    }
34}

117/// Builds the commerce `OpenAPI` document with title, version, and optional server.
118#[must_use]
119pub fn published_openapi(server_url: Option<&str>) -> utoipa::openapi::OpenApi {
120    let mut openapi = ApiDoc::openapi();
121    let description = openapi.info.description.clone();
122    let contact = openapi.info.contact.clone();
123    let license = openapi.info.license.clone();
124    serenade_openapi::finalize_openapi(
125        &mut openapi,
126        "rustashop-api",
127        env!("CARGO_PKG_VERSION"),
128        server_url,
129    );
130    openapi.info.description = description;
131    openapi.info.contact = contact;
132    openapi.info.license = license;
133    openapi
134}

152/// Registers explorer UIs (feature `openapi-ui`) on an Actix service config.
153#[cfg(feature = "openapi-ui")]
154pub fn configure_openapi_ui(cfg: &mut actix_web::web::ServiceConfig, server_url: Option<&str>) {
155    let openapi = published_openapi(server_url);
156    let paths = serenade_openapi::OpenApiUiPaths::new();
157    serenade_openapi::configure_actix_ui(cfg, openapi, &paths);
158}

175    fn published_openapi_sets_info() {
176        let doc = published_openapi(Some("http://127.0.0.1:8080"));
177        assert_eq!(doc.info.title, "rustashop-api");
178        assert_ne!(doc.info.version, "");
179        let servers = doc.servers.expect("servers");
180        assert_eq!(servers[0].url, "http://127.0.0.1:8080");
181    }

183    #[test]
184    fn openapi_uses_human_tags() {
185        let doc = ApiDoc::openapi();
186        let tags = doc.tags.expect("document tags");
187        let names: Vec<&str> = tags.iter().map(|tag| tag.name.as_str()).collect();
188        assert!(names.contains(&"products"));
189        assert!(names.contains(&"carts"));
190        assert!(names.iter().all(|name| !name.contains("::")));
191        let product_get = doc
192            .paths
193            .paths
194            .get("/v1/products")
195            .and_then(|item| item.get.as_ref())
196            .expect("GET /v1/products");
197        assert_eq!(
198            product_get.tags.as_deref(),
199            Some(["products".to_owned()].as_slice())
200        );
201    }
