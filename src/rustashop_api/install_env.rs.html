17/// Result of a successful install write.
18#[derive(Debug, Clone, PartialEq, Eq)]
19pub struct InstallWriteResult {
20    /// Opaque admin / API segment.
21    pub admin_prefix: String,
22    /// Bearer token written for operator API.
23    pub admin_token: String,
24    /// Absolute path to the env file updated.
25    pub env_path: PathBuf,
26}
27
28/// Errors from install env generation / write.
29#[derive(Debug, thiserror::Error)]
30pub enum InstallEnvError {
31    /// Overwrite refused without wipe acknowledgement.
32    #[error("admin prefix already set; pass wipe confirmation to overwrite")]
33    WipeRequired,
34    /// Invalid optional admin folder segment.
35    #[error("invalid admin folder: {0}")]
36    InvalidPrefix(String),
37    /// Filesystem failure.
38    #[error(transparent)]
39    Io(#[from] std::io::Error),
40}
41
42/// Options for [`run_install_write`].
43#[derive(Debug, Clone, Default)]
44pub struct InstallWriteOptions {
45    /// Optional explicit segment; otherwise random opaque.
46    pub admin_folder: Option<String>,
47    /// Required when `.env` already has a non-default admin prefix.
48    pub wipe_confirmed: bool,
49}

51/// Path to the env file used by install.
52#[must_use]
53pub fn env_file_path(root: &Path) -> PathBuf {
54    std::env::var_os(ENV_FILE_ENV).map_or_else(|| root.join(".env"), PathBuf::from)
55}
56
57/// True when an existing env already configures a non-default admin prefix.
58#[must_use]
59pub fn existing_prefix_needs_wipe(root: &Path) -> bool {
60    let path = env_file_path(root);
61    let Ok(contents) = fs::read_to_string(&path) else {
62        return false;
63    };
64    matches!(
65        read_env_value(&contents, ADMIN_API_PREFIX_ENV),
66        Some(value) if !value.is_empty() && value != DEFAULT_ADMIN_API_PREFIX
67    )
68}

70/// Generate segment + token and upsert into `.env`.
71///
72/// # Errors
73///
74/// Returns [`crate::InstallEnvError`] when wipe is required, the segment is invalid, or IO fails.
75pub fn run_install_write(
76    options: &InstallWriteOptions,
77) -> Result<InstallWriteResult, InstallEnvError> {
78    let root = shop_root();
79    if existing_prefix_needs_wipe(&root) && !options.wipe_confirmed {
80        return Err(InstallEnvError::WipeRequired);
81    }
82
83    let prefix = match &options.admin_folder {
84        Some(raw) => AdminApiPrefix::parse(raw).map_err(InstallEnvError::InvalidPrefix)?,
85        None => AdminApiPrefix::parse(&random_opaque_segment())
86            .map_err(InstallEnvError::InvalidPrefix)?,
87    };
88    let token = random_opaque_segment();
89    let path = env_file_path(&root);
90    upsert_env_file(
91        &path,
92        &[
93            (ADMIN_API_PREFIX_ENV, prefix.as_str()),
94            (ADMIN_TOKEN_ENV, token.as_str()),
95        ],
96    )?;
97
98    // Safety: ASCII keys; generated alphanumeric values.
99    unsafe {
100        std::env::set_var(ADMIN_API_PREFIX_ENV, prefix.as_str());
101        std::env::set_var(ADMIN_TOKEN_ENV, &token);
102    }
103
104    Ok(InstallWriteResult {
105        admin_prefix: prefix.as_str().to_owned(),
106        admin_token: token,
107        env_path: path,
108    })
109}

111/// Cryptographically opaque segment (`[a-z0-9]{20}`).
112#[must_use]
113pub fn random_opaque_segment() -> String {
114    let mut bytes = [0_u8; 20];
115    getrandom::fill(&mut bytes).expect("getrandom");
116    bytes
117        .iter()
118        .map(|b| {
119            const ALPHABET: &[u8] = b"abcdefghijklmnopqrstuvwxyz0123456789";
120            ALPHABET[usize::from(*b) % ALPHABET.len()] as char
121        })
122        .collect()
123}

125fn read_env_value(contents: &str, key: &str) -> Option<String> {
126    for line in contents.lines() {
127        let line = line.trim();
128        if line.is_empty() || line.starts_with('#') {
129            continue;
130        }
131        let Some((k, v)) = line.split_once('=') else {
132            continue;
133        };
134        if k.trim() == key {
135            return Some(v.trim().trim_matches('"').to_owned());
136        }
137    }
138    None
139}

141fn upsert_env_file(path: &Path, pairs: &[(&str, &str)]) -> Result<(), std::io::Error> {
142    let mut contents = if path.exists() {
143        fs::read_to_string(path)?
144    } else {
145        String::new()
146    };
147    if !contents.is_empty() && !contents.ends_with('\n') {
148        contents.push('\n');
149    }
150    for &(key, value) in pairs {
151        contents = upsert_key(&contents, key, value);
152    }
153    let parent = path.parent().unwrap_or_else(|| Path::new("."));
154    fs::create_dir_all(parent)?;
155    fs::write(path, contents)?;
156    Ok(())
157}
158
159fn upsert_key(contents: &str, key: &str, value: &str) -> String {
160    let mut out = String::new();
161    let mut replaced = false;
162    for line in contents.lines() {
163        let trimmed = line.trim();
164        let existing_key = trimmed.split_once('=').map(|(k, _)| k.trim());
165        if existing_key == Some(key) {
166            out.push_str(key);
167            out.push('=');
168            out.push_str(value);
169            out.push('\n');
170            replaced = true;
171            continue;
172        }
173        out.push_str(line);
174        out.push('\n');
175    }
176    if !replaced {
177        out.push_str(key);
178        out.push('=');
179        out.push_str(value);
180        out.push('\n');
181    }
182    out
183}

191    fn upsert_replaces_and_appends() {
192        let text = upsert_key("FOO=1\n", "BAR", "2");
193        assert!(text.contains("FOO=1"));
194        assert!(text.contains("BAR=2"));
195        let text2 = upsert_key(&text, "FOO", "9");
196        assert!(text2.contains("FOO=9"));
197        assert!(!text2.contains("FOO=1"));
