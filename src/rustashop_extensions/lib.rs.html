21/// Runs `adjust` on a `pricing-adjust` component at `component_path`.
22///
23/// # Errors
24///
25/// Returns an error when the component cannot be loaded or the guest call fails.
26pub fn invoke_pricing_adjust(
27    component_path: impl AsRef<Path>,
28    cart: &CartSnapshot,
29) -> Result<Vec<Adjustment>> {
30    let engine = Engine::default();
31    let component = Component::from_file(&engine, component_path.as_ref())
32        .with_context(|| format!("load component {}", component_path.as_ref().display()))?;
33    let linker = Linker::new(&engine);
34    let mut store = Store::new(&engine, ());
35    let instance = PricingAdjust::instantiate(&mut store, &component, &linker)
36        .context("instantiate pricing-adjust")?;
37    instance
38        .call_adjust(&mut store, cart)
39        .context("call adjust")
40}

42#[cfg(test)]
43mod tests {
44    use super::*;
45    use std::path::PathBuf;
46
47    fn fixture_component() -> PathBuf {
48        PathBuf::from(env!("CARGO_MANIFEST_DIR"))
49            .join("../../extensions/fixtures/pricing-adjust/pricing_adjust.component.wasm")
50    }
51
52    #[test]
53    fn pricing_adjust_applies_volume_discount() {
54        let cart = CartSnapshot {
55            currency: "EUR".into(),
56            lines: vec![CartLine {
57                sku: "HOODIE-M".into(),
58                quantity: 2,
59                unit_price: Money {
60                    amount_minor: 5000,
61                    currency: "EUR".into(),
62                },
63            }],
64        };
65        let adjustments = invoke_pricing_adjust(fixture_component(), &cart).expect("invoke");
66        assert_eq!(adjustments.len(), 1);
67        assert_eq!(adjustments[0].label, "volume-discount");
68        assert_eq!(adjustments[0].amount_minor, -1000);
69    }
70
71    #[test]
72    fn pricing_adjust_skips_small_carts() {
73        let cart = CartSnapshot {
74            currency: "EUR".into(),
